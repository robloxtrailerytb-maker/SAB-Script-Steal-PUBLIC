local function isAllowedPseudo()
	local list = CONFIG.ALLOWED_PSEUDOS
	if not list or #list == 0 then
		return true -- vide = tout le monde
	end
	local name = player.Name
	local dn = player.DisplayName
	for _, p in ipairs(list) do
		if p == name or p == dn then
			return true
		end
	end
	return false
end

local function makeDraggable(title, frame)
	local dragging, dragStart, startPos = false, nil, nil
	title.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			dragStart = input.Position
			startPos = frame.Position
		end
	end)
	title.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = false
		end
	end)
	UserInputService.InputChanged:Connect(function(input)
		if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
			local delta = input.Position - dragStart
			frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
		end
	end)
end

makeDraggable(settingsTitle, settingsFrame)
makeDraggable(animalsTitle, animalsFrame)

local function parseGeneration(text)
	if not text or text == "" then return 0 end
	text = text:gsub("/s", ""):gsub("%s+", ""):upper()
	local number, suffix = text:match("([%d%.]+)([KMBTQDN]*)")
	if not number then return 0 end
	number = tonumber(number) or 0
	local mult = { [""]=1, K=1e3, M=1e6, B=1e9, T=1e12, Q=1e15, QD=1e15, QN=1e18, D=1e33, N=1e30 }
	return number * (mult[suffix] or 1)
end

local function getBaseOwnerDisplayName(fromObj)
	local current = fromObj
	for _ = 1, 12 do
		if not current then break end
		local plotSign = current:FindFirstChild("PlotSign")
		if plotSign then
			local gui = plotSign:FindFirstChild("SurfaceGui")
			local frame = gui and gui:FindFirstChild("Frame")
			local label = frame and frame:FindFirstChild("TextLabel")
			if label and label:IsA("TextLabel") then
				local dn = (label.Text or ""):match("^(.+)'s Base$")
				if dn then return dn:gsub("^%s+", ""):gsub("%s+$", "") end
			end
		end
		current = current.Parent
	end
	return nil
end

local function isOwnedByLocalPlayer(obj)
	local dn = getBaseOwnerDisplayName(obj)
	return dn and (dn == player.DisplayName or dn == player.Name)
end

local function findModelsInPlots(name)
	local results = {}
	local plots = workspace:FindFirstChild("Plots")
	if not plots or not name then return results end
	local function recurse(parent)
		for _, child in ipairs(parent:GetChildren()) do
			if child:IsA("Model") and child.Name == name then
				table.insert(results, child)
			end
			if #child:GetChildren() > 0 then recurse(child) end
		end
	end
	recurse(plots)
	return results
end

local function getGenerationForName(name)
	local debris = workspace:FindFirstChild("Debris")
	if not debris then return 0, "-" end
	for _, template in ipairs(debris:GetChildren()) do
		if template.Name == "FastOverheadTemplate" then
			local ao = template:FindFirstChild("AnimalOverhead")
			if ao then
				local dn = ao:FindFirstChild("DisplayName")
				local gen = ao:FindFirstChild("Generation")
				if dn and dn:IsA("TextLabel") and dn.Text == name and gen and gen:IsA("TextLabel") then
					return parseGeneration(gen.Text), gen.Text
				end
			end
		end
	end
	return 0, "-"
end

local function findBestFromDebris()
	local debris = workspace:FindFirstChild("Debris")
	if not debris then return nil, nil, nil, 0 end

	-- Agrège par nom : on garde la meilleure gen pour chaque nom unique
	local bestByName = {}
	for _, template in ipairs(debris:GetChildren()) do
		if template.Name == "FastOverheadTemplate" then
			local ao = template:FindFirstChild("AnimalOverhead")
			if ao then
				local gen = ao:FindFirstChild("Generation")
				local dn = ao:FindFirstChild("DisplayName")
				if gen and gen:IsA("TextLabel") and dn and dn:IsA("TextLabel") then
					local name = dn.Text
					if name and name ~= "" then
						local val = parseGeneration(gen.Text)
						local prev = bestByName[name]
						if not prev or val > prev.Value then
							bestByName[name] = {Name = name, GenText = gen.Text, Value = val}
						end
					end
				end
			end
		end
	end

	local candidates = {}
	for _, c in pairs(bestByName) do
		table.insert(candidates, c)
	end
	table.sort(candidates, function(a, b) return a.Value > b.Value end)

	-- Premier animal qui a au moins un modèle en plot NON possédé par nous
	for _, c in ipairs(candidates) do
		local models = findModelsInPlots(c.Name)
		local stealable = {}
		for _, m in ipairs(models) do
			if m and m.Parent and not isOwnedByLocalPlayer(m) then
				table.insert(stealable, m)
			end
		end
		if #stealable > 0 then
			return c.Name, c.GenText, stealable, c.Value
		end
	end
	return nil, nil, nil, 0
end

			table.insert(currentAttachments, a1)

			beam.Attachment1 = a1

local function updateAll()
	local absName, absGenText, absModels, absValue = findBestFromDebris()

	-- Son vraiment à chaque changement de meilleur animal (nom différent)
	if absName and absName ~= lastBestName and absValue > 0 then
		pcall(function()
			newBestSound:Play()
		end)
	end
	if absValue > 0 then
		lastAbsoluteBestValue = math.max(lastAbsoluteBestValue, absValue)
	end

	local useName, useGenText, useModels = absName, absGenText, absModels

	if useName then
		nameLabel.Text = useName
		genLabel.Text = "Generation: " .. (useGenText or "-")
		-- Auto-active No Player Anims quand un meilleur animal (pseudo/nom) est affiché en haut
		if not noPlayerAnimsEnabled then
			setNoPlayerAnims(true)
		end
	else
		nameLabel.Text = "Aucun trouvé..."
		genLabel.Text = "Generation: -"
	end

	-- Toujours resynchroniser les modèles pour éviter un mauvais animal affiché
	if useName then
		local modelsOk = useModels and #useModels > 0 and useModels[1] and useModels[1].Parent
		if not modelsOk then
			useModels = findModelsInPlots(useName)
			local stealable = {}
			for _, m in ipairs(useModels or {}) do
				if m and m.Parent and not isOwnedByLocalPlayer(m) then
					table.insert(stealable, m)
				end
			end
			useModels = stealable
		end
	end

	if useName ~= lastBestName or useGenText ~= lastBestGenText then
		lastBestName = useName
		lastBestGenText = useGenText
		lastBestModels = useModels or {}
		updateVisuals(lastBestModels)
	elseif useName then
		lastBestModels = useModels or lastBestModels or {}
		updateVisuals(lastBestModels)
	end

	if showBaseOwnerEnabled and lastBestModels[1] then
		local dn = getBaseOwnerDisplayName(lastBestModels[1])
		if dn then
			local owner
			for _, plr in ipairs(Players:GetPlayers()) do
				if plr.DisplayName == dn or plr.Name == dn then owner = plr break end
			end
			if owner and owner.UserId ~= lastBaseOwnerUserId then
				applyBaseOwnerHighlight(owner)
			elseif not owner then
				clearBaseOwnerVisuals()
			end
		else
			clearBaseOwnerVisuals()
		end
	else
		clearBaseOwnerVisuals()
	end
end

local function setupViewport(viewport, model)
	for _, child in ipairs(viewport:GetChildren()) do
		child:Destroy()
	end

	local worldModel = Instance.new("WorldModel")
	worldModel.Parent = viewport

	local clone = model:Clone()
	for _, desc in ipairs(clone:GetDescendants()) do
		if desc:IsA("Script") or desc:IsA("LocalScript") or desc:IsA("Sound") or desc:IsA("ProximityPrompt") then
			desc:Destroy()
		end
	end
	clone.Parent = worldModel

	local cam = Instance.new("Camera")
	cam.Parent = viewport
	viewport.CurrentCamera = cam

	local ok, cf, size = pcall(function()
		return clone:GetBoundingBox()
	end)
	if ok and cf and size then
		local maxDim = math.max(size.X, size.Y, size.Z)
		cam.CFrame = cf * CFrame.new(0, size.Y * 0.15, maxDim * 1.6)
		cam.FieldOfView = 40
	else
		local part = clone.PrimaryPart or clone:FindFirstChildWhichIsA("BasePart")
		if part then
			cam.CFrame = part.CFrame * CFrame.new(0, 2, 8)
		end
	end
end

local function toggleFPSBoost(enable)
	if enable then
		savedData = {}

		-- 1) Supprime TOUS les MaterialVariant du jeu
		for _, obj in ipairs(workspace:GetDescendants()) do
			if obj:IsA("MaterialVariant") or obj.ClassName == "MaterialVariant" then
				table.insert(savedData, {Object = obj, Kind = "MaterialVariant", Parent = obj.Parent})
				pcall(function() obj.Parent = nil end)
			end
		end
		-- Aussi dans ReplicatedStorage / Lighting au cas où
		for _, serviceName in ipairs({"ReplicatedStorage", "Lighting", "MaterialService"}) do
			local svc = game:FindFirstChild(serviceName)
			if svc then
				for _, obj in ipairs(svc:GetDescendants()) do
					if obj:IsA("MaterialVariant") or obj.ClassName == "MaterialVariant" then
						table.insert(savedData, {Object = obj, Kind = "MaterialVariant", Parent = obj.Parent})
						pcall(function() obj.Parent = nil end)
					end
				end
			end
		end

		-- 2) Plots : material simple (pas de transparency forcée)
		local plots = workspace:FindFirstChild("Plots")
		if plots then
			for _, obj in ipairs(plots:GetDescendants()) do
				if obj:IsA("BasePart") then
					table.insert(savedData, {
						Object = obj,
						Kind = "Part",
						Material = obj.Material,
						CastShadow = obj.CastShadow,
						Reflectance = obj.Reflectance,
					})
					obj.Material = Enum.Material.SmoothPlastic
					obj.CastShadow = false
					obj.Reflectance = 0
				elseif obj:IsA("Texture") or obj:IsA("Decal") then
					table.insert(savedData, {Object = obj, Kind = "Texture", Texture = obj.Texture})
					obj.Texture = ""
				elseif obj:IsA("ParticleEmitter") or obj:IsA("Trail") or obj:IsA("Beam") then
					table.insert(savedData, {Object = obj, Kind = "Effect", Enabled = obj.Enabled})
					obj.Enabled = false
				end
			end
		end

		-- 3) Debris : allègement
		local debris = workspace:FindFirstChild("Debris")
		if debris then
			for _, obj in ipairs(debris:GetDescendants()) do
				if obj:IsA("BasePart") then
					table.insert(savedData, {
						Object = obj,
						Kind = "Part",
						Material = obj.Material,
						CastShadow = obj.CastShadow,
						Reflectance = obj.Reflectance,
					})
					obj.Material = Enum.Material.SmoothPlastic
					obj.CastShadow = false
					obj.Reflectance = 0
				elseif obj:IsA("Texture") or obj:IsA("Decal") then
					table.insert(savedData, {Object = obj, Kind = "Texture", Texture = obj.Texture})
					obj.Texture = ""
				end
			end
		end
	else
		for _, data in ipairs(savedData) do
			local obj = data.Object
			if obj then
				if data.Kind == "MaterialVariant" then
					if data.Parent and data.Parent.Parent then
						pcall(function() obj.Parent = data.Parent end)
					end
				elseif obj.Parent then
					if data.Kind == "Part" then
						obj.Material = data.Material
						obj.CastShadow = data.CastShadow
						obj.Reflectance = data.Reflectance or 0
					elseif data.Kind == "Texture" then
						obj.Texture = data.Texture
					elseif data.Kind == "Effect" then
						obj.Enabled = data.Enabled
					end
				end
			end
		end
		savedData = {}
	end
end

local function getRoot()
	local c = player.Character
	return c and (c:FindFirstChild("HumanoidRootPart") or c.PrimaryPart)
end

local function getJumpLaunchY(hum)
	if not hum then return 50 end
	if hum.UseJumpPower ~= false and tonumber(hum.JumpPower) and hum.JumpPower > 0 then
		return hum.JumpPower
	end
	local jh = tonumber(hum.JumpHeight)
	if jh and jh > 0 then return math.sqrt(2 * workspace.Gravity * jh) end
	return 50
end

local function launchHumanoid(hum)
	local root = getRoot()
	if not hum or not root then return end
	pcall(function()
		hum.Sit = false
		hum.PlatformStand = false
	end)
	local launchY = getJumpLaunchY(hum)
	local vel = root.AssemblyLinearVelocity
	root.AssemblyLinearVelocity = Vector3.new(vel.X, math.max(vel.Y, launchY), vel.Z)
end

local function controlledJump(hum, root, targetY)
	if not hum or not root then return end
	local diff = targetY - root.Position.Y
	if diff < 2 then
		local vel = root.AssemblyLinearVelocity
		if vel.Y > 10 then root.AssemblyLinearVelocity = Vector3.new(vel.X, vel.Y * 0.4, vel.Z) end
		return
	end
	local needed = math.clamp(diff * 1.8, 25, 55)
	local vel = root.AssemblyLinearVelocity
	root.AssemblyLinearVelocity = Vector3.new(vel.X, math.max(vel.Y, needed), vel.Z)
end

local function drawPath(waypoints)
	clearPathVisuals()
	if #waypoints < 2 then return end
	local folder = Instance.new("Folder")
	folder.Name = "BluePathVisual"
	folder.Parent = workspace
	table.insert(pathVisuals, folder)
	local step = math.max(1, math.floor(#waypoints / 8))
	for i = 1, #waypoints - 1, step do
		local nextI = math.min(i + step, #waypoints)
		local p1 = waypoints[i].Position + Vector3.new(0, 1.5, 0)
		local p2 = waypoints[nextI].Position + Vector3.new(0, 1.5, 0)
		local part = Instance.new("Part")
		part.Anchored = true
		part.CanCollide = false
		part.CanQuery = false
		part.CanTouch = false
		part.Material = Enum.Material.Neon
		part.Color = Color3.fromRGB(0, 170, 255)
		part.Transparency = 0.3
		part.Size = Vector3.new(0.5, 0.5, (p1 - p2).Magnitude)
		part.CFrame = CFrame.lookAt(p1, p2) * CFrame.new(0, 0, -part.Size.Z / 2)
		part.Parent = folder
		table.insert(pathVisuals, part)
	end
end

	if tick() - lastCloakTime < 4 then return end
	local tool = player.Backpack:FindFirstChild("Invisibility Cloak") or (character and character:FindFirstChild("Invisibility Cloak"))
	if tool then
		humanoid:EquipTool(tool)
		task.wait(0.1)
		tool:Activate()
		lastCloakTime = tick()
	end
end

-- ========== AUTO LOCK BASE ==========
local function findLocalPlotBlocks()
	if tick() - lastPlotCacheTime < 3 and #cachedPlotBlocks > 0 then
		local valid = {}
		for _, b in ipairs(cachedPlotBlocks) do
			if b and b.Parent then table.insert(valid, b) end
		end
		if #valid > 0 then return valid end
	end
	local results = {}
	local plots = workspace:FindFirstChild("Plots")
	if not plots then return results end
	for _, obj in ipairs(plots:GetDescendants()) do
		if obj.Name == "Purchases" then
			local owner = getBaseOwnerDisplayName(obj)
			if owner and (owner == player.DisplayName or owner == player.Name) then
				for _, child in ipairs(obj:GetChildren()) do
					if child.Name == "PlotBlock" and child:IsA("Model") then
						table.insert(results, child)
					end
				end
			end
		end
	end
	cachedPlotBlocks = results
	lastPlotCacheTime = tick()
	return results
end

local function getRemainingSeconds(plotBlock)
	local main = plotBlock:FindFirstChild("Main")
	local bb = main and main:FindFirstChild("BillboardGui")
	local rt = bb and bb:FindFirstChild("RemainingTime")
	if not rt then return nil end
	return tonumber(tostring(rt.Text or ""):match("(%d+)"))
end

local function getHitbox(plotBlock)
	return plotBlock:FindFirstChild("Hitbox")
		or (plotBlock:FindFirstChild("Main") and plotBlock.Main:FindFirstChild("Hitbox"))
		or plotBlock:FindFirstChildWhichIsA("BasePart")
end

local function getBlockY(plotBlock)
	local part = plotBlock:FindFirstChild("Main") or plotBlock:FindFirstChild("Hitbox") or plotBlock:FindFirstChildWhichIsA("BasePart")
	return part and part.Position.Y or math.huge
end

local function doLockBaseRush(hitbox)
	if isLockingBase or not hitbox or not humanoidRootPart then return end
	if tick() - lastLockBaseTime < 4 then return end
	isLockingBase = true
	lastLockBaseTime = tick()

	equipAndUseCloak()

	local path = PathfindingService:CreatePath({
		AgentRadius = 2.5, AgentHeight = 5, AgentCanJump = true, WaypointSpacing = 8,
	})
	local ok = pcall(function() path:ComputeAsync(humanoidRootPart.Position, hitbox.Position) end)
	local waypoints
	if ok and path.Status == Enum.PathStatus.Success then
		waypoints = path:GetWaypoints()
	else
		waypoints = {
			{Position = humanoidRootPart.Position, Action = Enum.PathWaypointAction.Walk},
			{Position = hitbox.Position, Action = Enum.PathWaypointAction.Walk},
		}
	end
	drawPath(waypoints)

	task.spawn(function()
		for _, wp in ipairs(waypoints) do
			if not autoLockBaseEnabled or not humanoidRootPart or not humanoid or humanoid.Health <= 0 then break end
			if wp.Action == Enum.PathWaypointAction.Jump then launchHumanoid(humanoid) end
			humanoid:MoveTo(wp.Position)
			local t0 = tick()
			while tick() - t0 < 2.5 do
				if not humanoidRootPart or not autoLockBaseEnabled then break end
				if (humanoidRootPart.Position - wp.Position).Magnitude < 5 then break end
				task.wait(0.12)
			end
		end
		if humanoid and hitbox and autoLockBaseEnabled then
			humanoid:MoveTo(hitbox.Position)
			task.wait(0.4)
		end
		task.wait(0.8)
		clearPathVisuals()
		isLockingBase = false
	end)
end

local function checkAutoLockBase()
	if not autoLockBaseEnabled or isLockingBase then return end
	if tick() - lastLockBaseTime < 3 then return end
	local blocks = findLocalPlotBlocks()
	if #blocks == 0 then return end
	local urgent = {}
	for _, b in ipairs(blocks) do
		local secs = getRemainingSeconds(b)
		if secs and secs <= 5 then table.insert(urgent, b) end
	end
	if #urgent == 0 then return end
	local lowest, lowestY = urgent[1], getBlockY(urgent[1])
	for i = 2, #urgent do
		local y = getBlockY(urgent[i])
		if y < lowestY then lowest, lowestY = urgent[i], y end
	end
	local hitbox = getHitbox(lowest)
	if hitbox then doLockBaseRush(hitbox) end
end

-- ========== AUTO FONCER ==========
local function doRush()
	if isRushing or isLockingBase or not lastBestName or not humanoidRootPart then return end
	if tick() - lastRushTime < 4 then return end
	isRushing = true
	lastRushTime = tick()

	local models = lastBestModels
	if not models or #models == 0 or not models[1] or not models[1].Parent then
		models = findModelsInPlots(lastBestName)
		lastBestModels = models
	end
	if #models == 0 then isRushing = false return end

	local targetModel = models[1]
	local targetPart = targetModel.PrimaryPart or targetModel:FindFirstChildWhichIsA("BasePart")
	if not targetPart then isRushing = false return end

	equipAndUseCloak()

	local needJump = (targetPart.Position.Y - humanoidRootPart.Position.Y) > 3.5
	local targetY = targetPart.Position.Y
	if needJump then setInfiniteJump(true) end

	local path = PathfindingService:CreatePath({
		AgentRadius = 2.5, AgentHeight = 5, AgentCanJump = true, WaypointSpacing = 8,
	})
	local ok = pcall(function() path:ComputeAsync(humanoidRootPart.Position, targetPart.Position) end)
	local waypoints
	if ok and path.Status == Enum.PathStatus.Success then
		waypoints = path:GetWaypoints()
	else
		waypoints = {
			{Position = humanoidRootPart.Position, Action = Enum.PathWaypointAction.Walk},
			{Position = targetPart.Position, Action = Enum.PathWaypointAction.Walk},
		}
	end
	drawPath(waypoints)

	task.spawn(function()
		for _, wp in ipairs(waypoints) do
			if not autoFoncerEnabled or not humanoidRootPart or not humanoid or humanoid.Health <= 0 then break end
			if wp.Action == Enum.PathWaypointAction.Jump or needJump then
				controlledJump(humanoid, humanoidRootPart, targetY)
			end
			humanoid:MoveTo(wp.Position)
			local t0 = tick()
			while tick() - t0 < 2.2 do
				if not humanoidRootPart or not autoFoncerEnabled then break end
				if (humanoidRootPart.Position - wp.Position).Magnitude < 5 then break end
				if needJump then controlledJump(humanoid, humanoidRootPart, targetY) end
				task.wait(0.12)
			end
		end
		if autoFoncerEnabled and humanoidRootPart then
			humanoid:MoveTo(targetPart.Position)
		end
		task.wait(0.8)
		clearPathVisuals()
		if needJump and not infiniteJumpEnabled then setInfiniteJump(false) end
		isRushing = false
	end)
end

local function checkStealNotification()
	if not autoKickEnabled then return end
	local notif = playerGui:FindFirstChild("Notification")
	local frame = notif and notif:FindFirstChild("Notification")
	if not frame then return end
	for _, child in ipairs(frame:GetChildren()) do
		if child.Name == "Template" and (child:IsA("TextLabel") or child:IsA("TextButton")) then
			if string.find(string.lower(child.Text or ""), "you stole") then
				player:Kick("Successful Steal!")
				return
			end
		end
	end
end

-- Boucle principale
task.spawn(function()
	local lastBestRefresh = 0
	while screenGui.Parent do
		local cam = workspace.CurrentCamera
		if cam and math.abs(cam.FieldOfView - customFOV) > 0.5 then
			cam.FieldOfView = customFOV
		end

		pcall(checkStealNotification)
		pcall(checkAutoLockBase)

		if autoFoncerEnabled and not isRushing and not isLockingBase then
			pcall(doRush)
		end

		if disableAnimsEnabled then
			pcall(disableAllAnimalAnimations)
		end

		if noPlayerAnimsEnabled then
			pcall(disablePlayerAnimations)
		end

		-- Refresh périodique du meilleur animal (évite le mauvais animal bloqué)
		if tick() - lastBestRefresh >= 3 then
			lastBestRefresh = tick()
			pcall(updateAll)
		end

		task.wait(1.2)
	end
end)
